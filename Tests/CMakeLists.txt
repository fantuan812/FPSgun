cmake_minimum_required(VERSION 3.16)
project(FPSAssemblyRulesTests LANGUAGES CXX)
add_executable(assembly_rules AssemblyRulesTests.cpp)
target_compile_features(assembly_rules PRIVATE cxx_std_17)
target_include_directories(assembly_rules PRIVATE ../Plugins/FPSAssembly/Source/FPSAssembly/Public)
if(MSVC)
  target_compile_options(assembly_rules PRIVATE /W4 /WX)
else()
  target_compile_options(assembly_rules PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME assembly_rules COMMAND assembly_rules)
